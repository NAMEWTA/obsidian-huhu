function streamWriter(res: ServerResponse) {
  const initialInput = mode === 'tools' ? 10 : 11;
  const send = (type: string, payload: Record<string, unknown> = {}) => {
    res.write(`event: ${type}\ndata: ${JSON.stringify({ type, ...payload })}\n\n`);
  };
  send('message_start', { message: {
    id: `msg_root_${requests.length}`, type: 'message', role: 'assistant', model: 'root-model', content: [],
    stop_reason: null, stop_sequence: null,
    usage: { input_tokens: initialInput - 6, cache_creation_input_tokens: 2, cache_read_input_tokens: 4, output_tokens: 0 }
  } });
  const text = (index: number, id: string, value: string, hang = false) => {
    const block = { type: 'text', text: value };
    send('content_block_start', { index, content_block: { type: 'text', text: '' } });
    send('content_block_delta', { index, delta: { type: 'text_delta', text: value } });
    if (!hang) send('content_block_stop', { index });
    return block;
  };
  const call = (index: number, block: any) => {
    send('content_block_start', { index, content_block: { ...block, input: {} } });
    const input = Array.from(JSON.stringify(block.input));
    const split = Math.floor(input.length / 2);
    send('content_block_delta', { index, delta: { type: 'input_json_delta', partial_json: input.slice(0, split).join('') } });
    send('content_block_delta', { index, delta: { type: 'input_json_delta', partial_json: input.slice(split).join('') } });
    send('content_block_stop', { index });
    return block;
  };
  const thinking = (index: number) => {
    const block = { type: 'thinking', thinking: '', signature: 'root-signed-empty-thinking-==' };
    send('content_block_start', { index, content_block: { type: 'thinking', thinking: '', signature: '' } });
    send('content_block_delta', { index, delta: { type: 'signature_delta', signature: block.signature } });
    send('content_block_stop', { index });
    return block;
  };
  const redacted = (index: number) => {
    const block = { type: 'redacted_thinking', data: 'opaque-root-保留-==' };
    send('content_block_start', { index, content_block: block });
    send('content_block_stop', { index });
    return block;
  };
  const finish = (blocks: any[], input: number, output: number) => {
    assert.equal(input, initialInput);
    send('message_delta', { delta: { stop_reason: null, stop_sequence: null }, usage: { output_tokens: output - 1 } });
    send('message_delta', { delta: { stop_reason: blocks.some(x => x.type === 'tool_use') ? 'tool_use' : 'end_turn', stop_sequence: null }, usage: { output_tokens: output } });
    send('message_stop'); res.end();
  };
  return { send, text, call, thinking, redacted, finish };
}

const pause = (ms: number) => new Promise<void>(resolve => setTimeout(resolve, ms));
async function until<T>(read: () => Promise<T>, predicate: (x: T) => boolean, label: string): Promise<T> {
  const deadline = Date.now() + 10_000; let last: T;
  do { last = await read(); if (predicate(last)) return last; await pause(10); } while (Date.now() < deadline);
  throw new Error(`${label}: ${JSON.stringify(last!)}`);
}
async function bounded<T>(work: Promise<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout>;
  try { return await Promise.race([work, new Promise<never>((_, reject) => { timer = setTimeout(() => reject(new Error(label)), 10_000); })]); }
  finally { clearTimeout(timer!); }
}
async function open() {

  // exit from the production transport's timeout termination during teardown.
  const child = (transport as unknown as { child: ChildProcess }).child;
  const childEnd = new Promise<void>(resolve => child.once('close', (code, signal) => { exits.push({ code, signal }); resolve(); }));

    const bot = await new HapConnection(transport, timers).connect({ name: 'root-anthropic-code-e2e', version: '1' });

  }
