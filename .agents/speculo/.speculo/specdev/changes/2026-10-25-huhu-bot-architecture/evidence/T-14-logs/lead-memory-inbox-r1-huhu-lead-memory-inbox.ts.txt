async function until<T>(read: () => Promise<T>, accept: (value: T) => boolean, label: string): Promise<T> {
  const deadline = Date.now() + 15000; let last: T | undefined;
  do { last = await read(); if (accept(last)) return last; if (serverError) throw new Error(serverError); await sleep(15); } while (Date.now() < deadline);
  throw new Error(`${label}: ${JSON.stringify(last)}`);
}
async function bounded<T>(work: Promise<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([work, new Promise<never>((_, reject) => { timer = setTimeout(() => reject(new Error(label)), 15000); })]); }
  finally { clearTimeout(timer); }
}
function texts(body: any): string[] {
  const content = (value: any): string => typeof value === 'string' ? value : Array.isArray(value) ? value.map(x => x.text ?? '').join('') : '';
  if (protocol === 'openai-responses') return [...(body.instructions ? [body.instructions] : []), ...(body.input ?? []).map((x: any) => content(x.content))];
  return [...(body.system ? [content(body.system)] : []), ...(body.messages ?? []).map((x: any) => content(x.content))];
}
function emit(res: ServerResponse, value: string) {
  res.setHeader('Content-Type', 'text/event-stream'); res.flushHeaders();
  if (protocol === 'openai-chat-completions') {
    const send = (payload: unknown) => res.write(`data: ${JSON.stringify(payload)}\n\n`);
    send({id: 'root-memory', choices: [{index: 0, delta: {role: 'assistant', content: value}, finish_reason: null}]});
    send({id: 'root-memory', choices: [{index: 0, delta: {}, finish_reason: 'stop'}]});
    send({choices: [], usage: {prompt_tokens: 11, completion_tokens: 5, total_tokens: 16}});
    res.end('data: [DONE]\n\n'); return;
  }
  if (protocol === 'anthropic-messages') {
    const send = (type: string, payload: object) => res.write(`event: ${type}\ndata: ${JSON.stringify({type, ...payload})}\n\n`);
    send('message_start', {message: {id: 'root-memory', type: 'message', role: 'assistant', model: 'root-model', content: [], stop_reason: null, stop_sequence: null, usage: {input_tokens: 11, cache_creation_input_tokens: 0, cache_read_input_tokens: 0, output_tokens: 0}}});
    send('content_block_start', {index: 0, content_block: {type: 'text', text: ''}});
    send('content_block_delta', {index: 0, delta: {type: 'text_delta', text: value}});
    send('content_block_stop', {index: 0});
    send('message_delta', {delta: {stop_reason: 'end_turn', stop_sequence: null}, usage: {output_tokens: 5}});
    send('message_stop', {}); res.end(); return;
  }
  let number = 0;
  const send = (type: string, payload: object) => res.write(`event: ${type}\ndata: ${JSON.stringify({type, sequence_number: number++, ...payload})}\n\n`);
  const envelope = (status: string, output: unknown[], usage?: object) => ({id: 'resp-root-memory', object: 'response', created_at: 1, model: 'root-model', status, output, error: null, incomplete_details: null, ...(usage ? {usage} : {})});
  const item = {id: 'msg-root-memory', type: 'message', role: 'assistant', status: 'completed', phase: 'final_answer', content: [{type: 'output_text', text: value, annotations: [], logprobs: []}]};
  send('response.created', {response: envelope('in_progress', [])});
  send('response.in_progress', {response: envelope('in_progress', [])});
  send('response.output_item.added', {output_index: 0, item: {...item, status: 'in_progress', content: []}});
  send('response.content_part.added', {item_id: item.id, output_index: 0, content_index: 0, part: {type: 'output_text', text: '', annotations: [], logprobs: []}});
  send('response.output_text.delta', {item_id: item.id, output_index: 0, content_index: 0, delta: value, logprobs: []});
  send('response.output_text.done', {item_id: item.id, output_index: 0, content_index: 0, text: value, logprobs: []});
  send('response.content_part.done', {item_id: item.id, output_index: 0, content_index: 0, part: item.content[0]});
  send('response.output_item.done', {output_index: 0, item});
  send('response.completed', {response: envelope('completed', [item], {input_tokens: 11, output_tokens: 5, total_tokens: 16, input_tokens_details: {cached_tokens: 0}, output_tokens_details: {reasoning_tokens: 0}})});
  res.end();
}
function toolTurn(res: ServerResponse, name: string, args: object, id: string) {
  res.setHeader('Content-Type', 'text/event-stream'); res.flushHeaders();
  const argumentsText = JSON.stringify(args);
  if (protocol === 'openai-chat-completions') {
    const send = (data: unknown) => res.write(`data: ${JSON.stringify(data)}\n\n`);
    send({id, choices: [{index: 0, delta: {role: 'assistant', tool_calls: [{index: 0, id, type: 'function', function: {name, arguments: argumentsText}}]}, finish_reason: null}]});
    send({id, choices: [{index: 0, delta: {}, finish_reason: 'tool_calls'}]});
    send({choices: [], usage: {prompt_tokens: 11, completion_tokens: 5, total_tokens: 16}});
    res.end('data: [DONE]\n\n'); return;
  }
  if (protocol === 'anthropic-messages') {
    const send = (type: string, data: object) => res.write(`event: ${type}\ndata: ${JSON.stringify({type, ...data})}\n\n`);
    send('message_start', {message: {id, type: 'message', role: 'assistant', model: 'root-model', content: [], stop_reason: null, stop_sequence: null, usage: {input_tokens: 11, output_tokens: 0}}});
    send('content_block_start', {index: 0, content_block: {type: 'tool_use', id, name, input: {}}});
    send('content_block_delta', {index: 0, delta: {type: 'input_json_delta', partial_json: argumentsText}});
    send('content_block_stop', {index: 0});
    send('message_delta', {delta: {stop_reason: 'tool_use', stop_sequence: null}, usage: {output_tokens: 5}});
    send('message_stop', {}); res.end(); return;
  }
  let sequence = 0;
  const send = (type: string, data: object) => res.write(`event: ${type}\ndata: ${JSON.stringify({type, sequence_number: sequence++, ...data})}\n\n`);
  const response = (status: string, output: object[]) => ({id: `resp_${id}`, object: 'response', created_at: 1, model: 'root-model', status, output, error: null, incomplete_details: null});
  const item = {id: `item_${id}`, type: 'function_call', call_id: id, name, arguments: '', status: 'in_progress'};
  send('response.created', {response: response('in_progress', [])});
  send('response.in_progress', {response: response('in_progress', [])});
  send('response.output_item.added', {output_index: 0, item});
  send('response.function_call_arguments.delta', {output_index: 0, item_id: item.id, delta: argumentsText});
  send('response.function_call_arguments.done', {output_index: 0, item_id: item.id, name, arguments: argumentsText});
  const final = {...item, arguments: argumentsText, status: 'completed'};
  send('response.output_item.done', {output_index: 0, item: final});
  send('response.completed', {response: {...response('completed', [final]), usage: {input_tokens: 11, output_tokens: 5, total_tokens: 16}}});
  res.end();
}

async function open() {
  const transport = await DesktopHapTransport.launch({binary, profileDir, cwd: root, credentialChannel: true, shutdownGraceMs: 5000,
    environment: {HOME: homeDir, USERPROFILE: homeDir, XDG_STATE_HOME: stateDir, LANG: 'C.UTF-8'}});
  const child = (transport as unknown as {child: ChildProcess}).child;
  const childEnd = new Promise<void>(resolve => child.once('close', (code, signal) => {exits.push({code, signal}); resolve();}));
  try {
    const bot = await new HapConnection(transport, timers).connect({name: 'root-memory-model', version: '1'});
    current = {bot, transport, childEnd}; return {...current, credentials: await transport.openCredentials(bot.hello.profile.profileId)};
  } catch (error) {transport.close(); await bounded(Promise.all([transport.exited, childEnd]), 'failed CLI open drain'); throw error;}
}
async function close() {
  if (!current) return; const active = current; current = undefined; active.bot.close();
  await bounded(Promise.all([active.transport.exited, active.childEnd]), 'CLI did not naturally drain');
  assert.deepEqual(exits.at(-1), {code: 0, signal: null}); assert.equal(active.transport.diagnostics.stderrBytes, 0);
}
