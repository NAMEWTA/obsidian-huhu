const pause = (ms = 5) => new Promise(resolve => setTimeout(resolve, ms));
async function until<T>(read: () => Promise<T>, accepts: (value: T) => boolean, label: string): Promise<T> {
	const deadline = Date.now() + 7000;
	let last: T;
	do { last = await read(); if (accepts(last)) return last; await pause(); } while (Date.now() < deadline);
	throw new Error(`${label}: ${JSON.stringify(last!)}`);
}

async function connect() {
	stderr = '';
	child = spawn(process.argv[2], ['-test.run=^TestLeadNoteRunner$'], { stdio: ['pipe', 'pipe', 'pipe'], env: { ...process.env, LEAD_NOTE_PROFILES: process.argv[3] } });
	const current = child; current.stderr.on('data', (chunk: Buffer) => { stderr += chunk; });
	ended = new Promise(resolve => current.once('exit', (code: number, signal: string) => resolve({ code, signal })));
	connection = new HapConnection({
		send: (data: Uint8Array) => new Promise<void>((resolve, reject) => current.stdin.write(data, (error: Error) => error ? reject(error) : resolve())),
		onData: (consume: (data: Uint8Array) => void) => { current.stdout.on('data', consume); return () => current.stdout.off('data', consume); },
		onClose: (consume: (error: Error) => void) => { const close = () => consume(new Error('closed')); current.on('exit', close); return () => current.off('exit', close); },
		close: () => current.stdin.end(),
	}, timers);
	bot = await connection.connect({ name: 'lead-note-e2e', version: '1' });
	host = await registerObsidianNoteHost(app, bot, { workspaceId: 'workspace:lead', originClientId: 'client:lead', timers, lifetime: { registerEvent() {} } });
	const implementation = host.host;
	bot.setHost({ ...implementation, invoke: async (request: unknown, signal: AbortSignal) => {
		const response = await implementation.invoke(request, signal);
		if (droppedResponse) { droppedResponse = false; throw new BotRpcError(-32000, { kind: 'HOST_UNAVAILABLE', reason: 'HOST_OFFLINE', retryable: false, correlationId: 'lead-loss' }); }
		return response;
	} });
	await bot.host.attach({ workspaceId: 'workspace:lead', capabilities: ['note.read', 'note.search', 'note.write'] });
}
async function disconnect() {
	if (!child) return;
	host?.dispose(); await host?.writes.drained(); bot?.close(); child.stdin.end();
	let forced = false; const timeout = setTimeout(() => { forced = true; child.kill('SIGKILL'); }, 4000);
	const exit = await ended; clearTimeout(timeout); child = undefined;
	assert.equal(forced, false, 'Runner required a forced kill'); assert.equal(exit.code, 0, JSON.stringify(exit)); assert.equal(stderr, '', 'Runner stderr');
	assert.equal(eventRefs.size, 0, 'Host left Vault listeners');
}

			operations: ['read', 'send', 'deliver'], sessionId: session.id }); grantIds = [grant.id];

}
