// This Lead-only probe reuses the controlled external Host/Engine fixtures from
// the existing contract suite, but independently drives the public APIs across
// a real Profile close/reopen after member removal. It never enters the repo.
type leadUnexpectedRecoveryEngine struct{ calls atomic.Int64 }

func (e *leadUnexpectedRecoveryEngine) Run(context.Context, api.EngineInput, func(api.EngineEvent) error) (api.EngineResult, error) {
	e.calls.Add(1)
	return api.EngineResult{}, fmt.Errorf("unexpected model restart")
}
func (e *leadUnexpectedRecoveryEngine) Resume(context.Context, api.EngineResumeInput, func(api.EngineEvent) error) (api.EngineResult, error) {
	e.calls.Add(1)
	return api.EngineResult{}, fmt.Errorf("unexpected model resume")
}

func TestLeadEffectOwnerAuditSurvivesRemovedMemberAndProfileReopen(t *testing.T) {
	root := filepath.Join(t.TempDir(), "profiles")
	if err := os.Mkdir(root, 0700); err != nil {
		t.Fatal(err)
	}
	locator := filepath.Join(root, "one")
	c, err := bootstrap.OpenWithOptions(ctx, root, locator, bootstrap.Options{Engine: engine{}})
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		if c != nil {
			if err := c.Close(); err != nil {
				t.Error(err)
			}
		}
	})
	h := &host{queryStatus: "reconciled"}
	h.lease, err = c.API.Host.Attach(ctx, identity, api.HostAttachCommand{WorkspaceID: "vault", Capabilities: []string{"note.write"}}, h)
	if err != nil {
		t.Fatal(err)
	}
	a, err := c.API.Assistant.Create(ctx, identity, api.AgentCreate{CommandMeta: meta(t, c, "lead-a", ""), Draft: api.AgentDraft{Name: "A", Persona: "coordinator", AllowedTools: []string{}}})
	if err != nil {
		t.Fatal(err)
	}
	b, err := c.API.Assistant.Create(ctx, identity, api.AgentCreate{CommandMeta: meta(t, c, "lead-b", ""), Draft: api.AgentDraft{Name: "B", Persona: "writer", AllowedTools: []string{"note.write"}}})
	if err != nil {
		t.Fatal(err)
	}
	session, err := c.API.Groups.CreateSession(ctx, identity, api.SessionCreate{CommandMeta: meta(t, c, "lead-session", ""), Kind: "group", Title: "owner audit after restart", CoordinatorID: a.ID, MemberIDs: []string{a.ID, b.ID}})
	if err != nil {
		t.Fatal(err)
	}
	accepted, err := c.API.Groups.StartRun(ctx, identity, api.RunStart{CommandMeta: meta(t, c, "lead-start", session.Revision), SessionID: session.ID, Intent: "write", SourceRefs: []api.SourceSnapshot{}, TargetAgentIDs: []string{b.ID}})
	if err != nil || len(accepted.TargetRunIDs) != 1 {
		t.Fatalf("accept %#v %v", accepted, err)
	}
	profile := receiptProfile(t, c)
	child := accepted.TargetRunIDs[0]
	var approval api.Approval
	await(t, func() bool {
		p, e := c.API.Approvals.List(ctx, identity, api.PageQuery{ProfileID: profile})
		if e != nil {
			t.Fatal(e)
		}
		for _, a := range p.Items {
			if a.RunID == child {
				approval = a
				return true
			}
		}
		return false
	})
	_, err = c.API.Approvals.Resolve(ctx, identity, api.ApprovalResolve{CommandMeta: meta(t, c, "lead-approve", approval.Revision), ID: approval.ID, EffectID: approval.EffectID, Decision: "approve", PreviewHash: approval.Preview.PreviewHash})
	if err != nil {
		t.Fatal(err)
	}
	await(t, func() bool {
		r, e := c.API.Conversation.GetRun(ctx, identity, api.IDQuery{ProfileID: profile, ID: child})
		return e == nil && r.Status == "failed" && r.EffectsUnresolved
	})
	await(t, func() bool {
		r, e := c.API.Conversation.GetRun(ctx, identity, api.IDQuery{ProfileID: profile, ID: accepted.RunID})
		return e == nil && (r.Status == "failed" || r.Status == "succeeded")
	})
	session, err = c.API.Conversation.GetSession(ctx, identity, api.IDQuery{ProfileID: profile, ID: session.ID})
	if err != nil {
		t.Fatal(err)
	}
	_, err = c.API.Groups.UpdateMembers(ctx, identity, api.SessionMembersUpdate{CommandMeta: meta(t, c, "lead-remove", session.Revision), ID: session.ID, MemberIDs: []string{a.ID}, HistoryScope: api.GroupHistoryScope{Kind: "none"}})
	if err != nil {
		t.Fatal(err)
	}
	before, err := c.API.Effects.Get(ctx, identity, api.EffectQuery{ProfileID: profile, ID: approval.EffectID})
	if err != nil || before.Status != "unknown_outcome" {
		t.Fatalf("owner unknown after removal %#v %v", before, err)
	}
	if err = c.Close(); err != nil {
		t.Fatal(err)
	}
	c = nil
	recoveredEngine := &leadUnexpectedRecoveryEngine{}
	c, err = bootstrap.OpenWithOptions(ctx, root, locator, bootstrap.Options{Engine: recoveredEngine})
	if err != nil {
		t.Fatal(err)
	}
	persisted, err := c.API.Effects.Get(ctx, identity, api.EffectQuery{ProfileID: profile, ID: approval.EffectID})
	if err != nil || !reflect.DeepEqual(persisted, before) {
		t.Fatalf("unknown changed across Profile reopen: %#v %#v %v", before, persisted, err)
	}
	if _, err = c.API.Approvals.Get(ctx, identity, api.IDQuery{ProfileID: profile, ID: approval.ID}); err == nil {
		t.Fatal("removed member approval Preview exposed")
	}
	foreign := api.NewConnectionIdentity("lead-foreign", "not-the-owner", true)
	if _, err = c.API.Effects.Get(ctx, foreign, api.EffectQuery{ProfileID: profile, ID: approval.EffectID}); err == nil {
		t.Fatal("foreign principal read effect")
	}
	if _, err = c.API.Effects.Get(ctx, identity, api.EffectQuery{ProfileID: "foreign-profile", ID: approval.EffectID}); err == nil {
		t.Fatal("foreign Profile read effect")
	}
	command := api.EffectReconcile{CommandMeta: meta(t, c, "lead-reconcile", persisted.Revision), ID: approval.EffectID}
	if _, err = c.API.Effects.Reconcile(ctx, foreign, command); err == nil {
		t.Fatal("foreign principal reconciled")
	}
	if _, err = c.API.Effects.Reconcile(ctx, identity, command); err == nil {
		t.Fatal("reconcile without current Host unexpectedly succeeded")
	}
	h.mu.Lock()
	w, q := h.writes, h.queries
	h.mu.Unlock()
	if w != 1 || q != 0 {
		t.Fatalf("denied paths called Host %d/%d", w, q)
	}
	h.lease, err = c.API.Host.Attach(ctx, identity, api.HostAttachCommand{WorkspaceID: "vault", Capabilities: []string{"note.write"}}, h)
	if err != nil {
		t.Fatal(err)
	}
	view, err := c.API.Effects.Reconcile(ctx, identity, command)
	if err != nil || view.Status != "reconciled" {
		t.Fatalf("owner reconciliation %#v %v", view, err)
	}
	duplicate, err := c.API.Effects.Reconcile(ctx, identity, command)
	if err != nil || !reflect.DeepEqual(view, duplicate) {
		t.Fatalf("exact retry changed result %#v %#v %v", view, duplicate, err)
	}
	stale := command
	stale.CommandID = "lead-stale"
	stale.IdempotencyKey = "lead-stale"
	if _, err = c.API.Effects.Reconcile(ctx, identity, stale); err == nil {
		t.Fatal("stale Run revision accepted")
	}
	body, err := json.Marshal(view)
	if err != nil {
		t.Fatal(err)
	}
	for _, forbidden := range []string{"new exact", "opaque-target", "old -> new", "\"preview\"", "\"content\"", "\"target\""} {
		if strings.Contains(string(body), forbidden) {
			t.Fatalf("body-bearing field in EffectView: %s", forbidden)
		}
	}
	h.mu.Lock()
	w, q = h.writes, h.queries
	h.mu.Unlock()
	if w != 1 || q != 1 {
		t.Fatalf("expected one original write and one owner receipt query: %d/%d", w, q)
	}
	if err = c.Close(); err != nil {
		t.Fatal(err)
	}
	c = nil
	if recoveredEngine.calls.Load() != 0 {
		t.Fatalf("model restarted %d times", recoveredEngine.calls.Load())
	}
	db, err := sql.Open("sqlite3", "file:"+filepath.Join(locator, "profile.sqlite3")+"?mode=ro")
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	var status string
	var unresolved, pins int
	if err = db.QueryRow("SELECT status,effects_unresolved FROM execution_runs WHERE run_id=?", child).Scan(&status, &unresolved); err != nil {
		t.Fatal(err)
	}
	if err = db.QueryRow("SELECT (SELECT count(*) FROM config_snapshot_pins)+(SELECT count(*) FROM task_config_pins)").Scan(&pins); err != nil {
		t.Fatal(err)
	}
	if status != "failed" || unresolved != 0 || pins != 0 {
		t.Fatalf("terminal/pin state %s/%d/%d", status, unresolved, pins)
	}
	t.Logf("Profile reopened; owner reconciled retained effect, removed Preview/foreign/no Host refused; original Invoke=%d QueryReceipt=%d restarted models=%d terminal=%s unresolved=%d pins=%d", w, q, recoveredEngine.calls.Load(), status, unresolved, pins)
}
