async function until<T>(read: () => Promise<T>, accepts: (value: T) => boolean, label: string): Promise<T> {
	const deadline = Date.now() + 7000;
	let last: T;
	do { last = await read(); if (accepts(last)) return last; await pause(); } while (Date.now() < deadline);
	throw new Error(`${label}: ${JSON.stringify(last!)}`);
}

async function connect() {
	stderr = '';

	connection = new HapConnection({
		send: (data: Uint8Array) => new Promise<void>((resolve, reject) => current.stdin.write(data, (error: Error) => error ? reject(error) : resolve())),
		onData: (consume: (data: Uint8Array) => void) => { current.stdout.on('data', consume); return () => current.stdout.off('data', consume); },
		onClose: (consume: (error: Error) => void) => { const close = () => consume(new Error('closed')); current.on('exit', close); return () => current.off('exit', close); },
		close: () => current.stdin.end(),

	bot = await connection.connect({ name: 'lead-groups-e2e', version: '1' });
	host = await registerObsidianNoteHost(app, bot, { workspaceId: 'workspace:lead', originClientId: 'client:lead', timers, lifetime: { registerEvent() {} } });

		if (droppedResponse) { droppedResponse = false; throw new BotRpcError(-32000, { kind: 'HOST_UNAVAILABLE', reason: 'HOST_OFFLINE', retryable: false, correlationId: 'lead-loss' }); }
