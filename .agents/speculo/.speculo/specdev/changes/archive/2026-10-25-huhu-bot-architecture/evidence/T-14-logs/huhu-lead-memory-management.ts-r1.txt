const timers = new Set<ReturnType<typeof setTimeout>>();
const timerPort = { after(ms: number, fn: () => void) {
  const timer = setTimeout(() => { timers.delete(timer); fn(); }, ms);
  timers.add(timer); return () => { clearTimeout(timer); timers.delete(timer); };
} };
let current: { transport: DesktopHapTransport; bot: BotClient; childEnd: Promise<void> } | undefined;

async function bounded<T>(work: Promise<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([work, new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(label)), 15_000);
  })]); } finally { clearTimeout(timer); }
}
async function open() {
  const transport = await DesktopHapTransport.launch({ binary, profileDir, cwd: root,
    shutdownGraceMs: 5000, environment: { HOME: home, USERPROFILE: home, XDG_STATE_HOME: state, LANG: 'C.UTF-8' } });
  const child = (transport as unknown as { child: ChildProcess }).child;
  const childEnd = new Promise<void>(resolve => child.once('close', (code, signal) => {
    exits.push({ code, signal }); resolve();
  }));
  try {
    const bot = await new HapConnection(transport, timerPort).connect({ name: 'root-memory-management', version: '1' });
    current = { transport, bot, childEnd }; return bot;
  } catch (error) {
    transport.close(); await bounded(Promise.all([transport.exited, childEnd]), 'failed open did not drain'); throw error;
  }
}
async function close() {
  if (!current) return;
  const active = current; current = undefined;
  active.bot.close();
  await bounded(Promise.all([active.transport.exited, active.childEnd]), 'CLI failed to drain');
  assert.deepEqual(exits.at(-1), { code: 0, signal: null });
  assert.equal(active.transport.diagnostics.stderrBytes, 0);
}

const hash = (text: string) => 'sha256:' + createHash('sha256').update(text, 'utf8').digest('hex');
async function main() {

  let bot = await open(); const profileId = bot.hello.profile.profileId;

  const na: MemoryNamespace = { kind: 'agent', id: a.id }, nb: MemoryNamespace = { kind: 'agent', id: b.id };
