type eventEngine struct {
	database string
	release  chan struct{}
	once     sync.Once
}

func (e *eventEngine) Run(ctx context.Context, in api.EngineInput, emit func(api.EngineEvent) error) (api.EngineResult, error) {
	if strings.HasPrefix(in.Intent, "approval") {
		result, err := in.Tools.Invoke(ctx, api.ToolCall{ID: "call:events", Name: "note.write", Target: api.EffectTarget{WorkspaceID: "workspace:events", ResourceID: "note:" + in.RunID, Operation: "update", BaseRevision: "1"}, Content: "proposed body"})
		if err != nil {
			return api.EngineResult{}, err
		}
		if result.Status != "approval_required" {
			return api.EngineResult{}, fmt.Errorf("expected actual Broker suspension")
		}
		checkpoint, err := in.Checkpoints.Publish(ctx, bytes.NewBufferString("events-private-checkpoint"))
		if err != nil {
			return api.EngineResult{}, err
		}
		return api.EngineResult{Status: "suspended", Suspension: &api.EngineSuspension{Token: result.SuspensionToken, Checkpoint: checkpoint}}, nil
	}
	if in.Intent == "hold" {
		if err := emit(api.EngineEvent{Type: "delta", Text: "not canceled by page disposal"}); err != nil {
			return api.EngineResult{}, err
		}
		select {
		case <-e.release:
			return api.EngineResult{Status: "finished", Text: "continued"}, nil
		case <-ctx.Done():
			return api.EngineResult{}, ctx.Err()
		}
	}
	if in.Intent == "release" {
		e.once.Do(func() { close(e.release) })
		return api.EngineResult{Status: "finished", Text: "released"}, nil
	}
	if strings.HasPrefix(in.Intent, "late:") || in.Intent == "big" {
		db, err := sql.Open("sqlite3", "file:"+filepath.ToSlash(e.database)+"?_pragma=busy_timeout(500)")
		if err != nil {
			return api.EngineResult{}, err
		}
		defer db.Close()
		tx, err := db.BeginTx(ctx, nil)
		if err != nil {
			return api.EngineResult{}, err
		}
		defer tx.Rollback()
		if in.Intent == "big" {
			if _, err = tx.ExecContext(ctx, "DELETE FROM execution_events WHERE run_id=?", in.RunID); err == nil {
				_, err = tx.ExecContext(ctx, "UPDATE execution_runs SET event_seq=9007199254740992 WHERE run_id=?", in.RunID)
			}
		} else {
			err = executionstore.AddEventInTx(ctx, tx, strings.TrimPrefix(in.Intent, "late:"), "reconciled", map[string]string{"effectId": "effect:late", "status": "succeeded"})
		}
		if err != nil {
			return api.EngineResult{}, err
		}
		if err = tx.Commit(); err != nil {
			return api.EngineResult{}, err
		}
	}
	if in.Intent == "private" {
		if len(in.Context.Sources) != 1 || in.Context.Sources[0].Text != "PRIVATE_SOURCE" {
			return api.EngineResult{}, fmt.Errorf("actual explicit Host source missing")
		}
		if err := emit(api.EngineEvent{Type: "delta", Text: "DERIVED_PRIVATE"}); err != nil {
			return api.EngineResult{}, err
		}
		if err := emit(api.EngineEvent{Type: "tool.fixture", Text: "PRIVATE_LEDGER"}); err != nil {
			return api.EngineResult{}, err
		}
		return api.EngineResult{Status: "finished", Text: "DERIVED_PRIVATE"}, nil
	}
	if strings.HasPrefix(in.Intent, "large:") {
		text := strings.Repeat("a", 65536)
		switch in.Intent {
		case "large:control":
			text = strings.Repeat("a", 65000)
		case "large:escaped":
			text = strings.Repeat("\x00\"\\", 21845) + "\x00"
		case "large:source":
			if len(in.Context.Sources) != 1 || in.Context.Sources[0].Text != "PRIVATE_SOURCE" {
				return api.EngineResult{}, fmt.Errorf("large event lost actual Host source")
			}
		}
		if err := emit(api.EngineEvent{Type: "text.delta", Text: text}); err != nil {
			return api.EngineResult{}, err
		}
		return api.EngineResult{Status: "finished", Text: text}, nil
	}
	count, text := 1, "甲😀\n乙"
	if in.Intent == "compact" {
		count, text = 270, "x"
	}
	debug086Start:=time.Now()
	if in.Intent=="compact" {debugbase.Debug086Reset();defer func(){fmt.Fprintf(os.Stderr,"[DEBUG-086] compact emits elapsed=%s metrics=%#v\n",time.Since(debug086Start),debugbase.Debug086Snapshot());close(debug086CompactDone)}()}
	for range count {
		if err := emit(api.EngineEvent{Type: "delta", Text: text}); err != nil {
			return api.EngineResult{}, err
		}
	}
	return api.EngineResult{Status: "finished", Text: strings.Repeat(text, count)}, nil
}
func (*eventEngine) Resume(ctx context.Context, in api.EngineResumeInput, _ func(api.EngineEvent) error) (api.EngineResult, error) {
	reader, err := in.Input.Checkpoints.Open(ctx, in.Checkpoint)
	if err != nil {
		return api.EngineResult{}, err
	}
	data, err := io.ReadAll(reader)
	closeErr := reader.Close()
	if err != nil || closeErr != nil || string(data) != "events-private-checkpoint" {
		return api.EngineResult{}, fmt.Errorf("actual checkpoint failed validation")
	}
	return api.EngineResult{Status: "finished", Text: "decision " + in.Decision.Status}, nil
}

type eventNoteWriter struct{ host api.HostAPI }

func (w *eventNoteWriter) Preview(ctx context.Context, target api.EffectTarget, text string) (api.NoteTargetPreview, error) {
	lease, err := w.host.CurrentLease(ctx)
	if err != nil {
		return api.NoteTargetPreview{}, err
	}
	sum := sha256.Sum256([]byte(text))
	return api.NoteTargetPreview{Target: target, Tool: brokerapp.Descriptor, ContentHash: "sha256:" + hex.EncodeToString(sum[:]), Diff: "fixture target -> proposed body", Lease: lease}, nil
}

func (*eventNoteWriter) Invoke(_ context.Context, dispatch api.EffectDispatch) (api.HostReceipt, error) {
	return api.HostReceipt{EffectID: dispatch.EffectID, Status: "committed", Revision: "2", ReceiptID: "receipt:fixture"}, nil
}
func TestEventsRunnerProcess(t *testing.T) {
	index := -1
	for i, arg := range os.Args {
		if arg == "--events-runner" {
			index = i
		}
	}
	if index < 0 {
		return
	}
	if index+1 >= len(os.Args) {
		os.Exit(2)
	}
	root := os.Args[index+1]
	runCtx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	defer cancel()
	engine := &eventEngine{database: filepath.Join(root, "one", "profile.sqlite3"), release: make(chan struct{})}
	writer := &eventNoteWriter{}
	core, e := bootstrap.OpenWithOptions(runCtx, root, filepath.Join(root, "one"), bootstrap.Options{Engine: engine, NoteWriter: writer})
	if e != nil {
		fmt.Fprintln(os.Stderr, e)
		os.Exit(1)
	}
	writer.host = core.API.Host
	server := stdio.New(os.Stdin, os.Stdout, core.API, api.NewConnectionIdentity("stdio", "user", true))
	e = server.Run(runCtx)
	server.Close()
	closeErr := core.Close()
	if e != nil || closeErr != nil {
		fmt.Fprintln(os.Stderr, e, closeErr)
		os.Exit(1)
	}
	os.Exit(0)
}

func TestActualTSClientConsumesDurableGoEventsSnapshotsAndLateFacts(t *testing.T) {
	repo, e := filepath.Abs(filepath.Join("..", "..", "..", "..", ".."))
	if e != nil {
		t.Fatal(e)
	}
	root := t.TempDir()
	profiles := filepath.Join(root, "profiles")
	if e = os.Mkdir(profiles, 0700); e != nil {
		t.Fatal(e)
	}
	source, bundle := filepath.Join(root, "events.ts"), filepath.Join(root, "events.cjs")
	if e = os.WriteFile(source, []byte(strings.ReplaceAll(actualEventFixture, "__REPO__", filepath.ToSlash(repo))), 0600); e != nil {
		t.Fatal(e)
	}
	buildCtx, buildCancel := context.WithTimeout(ctx, time.Minute)
	defer buildCancel()
	build := exec.CommandContext(buildCtx, "pnpm", "exec", "esbuild", source, "--bundle", "--platform=node", "--format=cjs", "--outfile="+bundle)
	build.Dir = repo
	if output, e := build.CombinedOutput(); e != nil {
		t.Fatalf("actual EventClient bundle: %v %s", e, output)
	}
	runCtx, cancel := context.WithTimeout(ctx, 25*time.Second)
	defer cancel()
	cmd := exec.CommandContext(runCtx, "node", bundle, os.Args[0], profiles)
	var output, stderr bytes.Buffer
	cmd.Stdout = &output
	cmd.Stderr = &stderr
	if e = cmd.Run(); e != nil {
		t.Fatalf("actual Go↔TS events: %v\n%s\n%s", e, output.String(), stderr.String())
	}
	var observed map[string]bool
	if stderr.Len() != 0 || cmd.ProcessState == nil || json.Unmarshal(output.Bytes(), &observed) != nil || len(observed) != 15 {
		t.Fatal("invalid actual observations/cleanup", output.String(), stderr.String())
	}
	for k, v := range observed {
		if !v {
			t.Fatal(k)
		}
	}
	t.Logf("production EventClient/reducer/HapConnection ↔ Go bootstrap/stdio/SQLite; fake Engine + explicit large-seq/late-ledger SQL fixture only: %s; Node and Runner Wait/exit", strings.TrimSpace(output.String()))
}
